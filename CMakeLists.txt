cmake_minimum_required(VERSION 3.20)
project(PowerDriveSim VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(powerdrive core/model/model.cpp core/model/waveform.cpp core/model/connectivity.cpp core/model/hierarchy.cpp core/model/units.cpp core/editor/document.cpp core/editor/properties.cpp formats/project/project.cpp
  core/model/semiconductor.cpp core/editor/hierarchy.cpp core/compiler/compiler.cpp core/compiler/topology.cpp core/solver/reference/reference.cpp core/solver/reference/factorization.cpp core/solver/reference/equation_cache.cpp formats/samples/table.cpp results/csv.cpp results/measurements.cpp)
target_include_directories(powerdrive PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Threads REQUIRED)
target_link_libraries(powerdrive PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(powerdrive PUBLIC /W4 /permissive- /utf-8 /fp:strict)
else()
  target_compile_options(powerdrive PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(powerdrive PUBLIC -ffp-contract=off)
endif()
add_executable(powerdrive-cli apps/cli/main.cpp)
target_link_libraries(powerdrive-cli PRIVATE powerdrive)

option(PDS_BUILD_DESKTOP "Build Qt 6 desktop editor" OFF)
if(PDS_BUILD_DESKTOP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent)
  add_library(powerdrive-desktop-ui apps/desktop/editor.cpp apps/desktop/theme.cpp apps/desktop/hierarchy.cpp apps/desktop/actions.cpp apps/desktop/plots.cpp apps/desktop/views.cpp apps/desktop/scope_navigation.cpp apps/desktop/scope_measurements.cpp apps/desktop/scope_style.cpp apps/desktop/labels.cpp apps/desktop/canvas.cpp apps/desktop/routing.cpp apps/desktop/inspector.cpp apps/desktop/palette.cpp apps/desktop/property_schema.cpp apps/desktop/property_fields.cpp apps/desktop/translations.qrc)
  set_target_properties(powerdrive-desktop-ui PROPERTIES AUTORCC ON)
  target_link_libraries(powerdrive-desktop-ui PUBLIC powerdrive Qt6::Widgets Qt6::Concurrent)
  add_executable(powerdrive-desktop WIN32 apps/desktop/main.cpp)
  if(WIN32)
    target_sources(powerdrive-desktop PRIVATE apps/desktop/app.rc)
  endif()
  target_link_libraries(powerdrive-desktop PRIVATE powerdrive-desktop-ui)
  add_custom_command(TARGET powerdrive-desktop POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/examples $<TARGET_FILE_DIR:powerdrive-desktop>/examples
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/apps/desktop/components $<TARGET_FILE_DIR:powerdrive-desktop>/components
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/library $<TARGET_FILE_DIR:powerdrive-desktop>/library)
endif()
if(PDS_BUILD_DESKTOP AND BUILD_TESTING)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
  add_executable(powerdrive-desktop-tests tests/desktop_tests.cpp)
  set_target_properties(powerdrive-desktop-tests PROPERTIES AUTOMOC ON)
  target_compile_definitions(powerdrive-desktop-tests PRIVATE PDS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(powerdrive-desktop-tests PRIVATE powerdrive-desktop-ui Qt6::Test)
  add_test(NAME desktop COMMAND powerdrive-desktop-tests -platform offscreen -o -,txt)
  add_executable(powerdrive-interaction-tests tests/interaction_tests.cpp)
  set_target_properties(powerdrive-interaction-tests PROPERTIES AUTOMOC ON)
  target_compile_definitions(powerdrive-interaction-tests PRIVATE PDS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(powerdrive-interaction-tests PRIVATE powerdrive-desktop-ui Qt6::Test)
  if(WIN32)
    target_link_libraries(powerdrive-interaction-tests PRIVATE psapi)
  endif()
  add_test(NAME desktop-interaction COMMAND powerdrive-interaction-tests -platform offscreen -o -,txt)
  add_test(NAME desktop-launch COMMAND powerdrive-desktop --smoke-test ${CMAKE_CURRENT_SOURCE_DIR}/examples/rc.pds -platform offscreen)
  set_tests_properties(desktop desktop-launch desktop-interaction PROPERTIES TIMEOUT 45)
  if(TARGET Qt6::QOffscreenIntegrationPlugin)
    set_tests_properties(desktop desktop-launch desktop-interaction PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM_PLUGIN_PATH=$<TARGET_FILE_DIR:Qt6::QOffscreenIntegrationPlugin>")
  endif()
  if(WIN32)
    file(TO_CMAKE_PATH "$ENV{WINDIR}/Fonts" PDS_TEST_FONT_DIRECTORY)
    set_property(TEST desktop desktop-launch desktop-interaction APPEND PROPERTY
      ENVIRONMENT "QT_QPA_FONTDIR=${PDS_TEST_FONT_DIRECTORY}")
  endif()
endif()
