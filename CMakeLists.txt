cmake_minimum_required(VERSION 3.20)
project(PowerDriveSim VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(powerdrive core/model/model.cpp formats/project/project.cpp
  core/compiler/compiler.cpp core/solver/reference/reference.cpp results/csv.cpp)
target_include_directories(powerdrive PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(powerdrive PUBLIC /W4 /permissive- /utf-8 /fp:strict)
else()
  target_compile_options(powerdrive PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(powerdrive PUBLIC -ffp-contract=off)
endif()
add_executable(powerdrive-cli apps/cli/main.cpp)
target_link_libraries(powerdrive-cli PRIVATE powerdrive)
include(CTest)
if(BUILD_TESTING)
  add_executable(powerdrive-tests tests/tests.cpp)
  target_link_libraries(powerdrive-tests PRIVATE powerdrive)
  foreach(group unit numerical serialization topology regression examples)
    add_test(NAME ${group} COMMAND powerdrive-tests ${group} ${CMAKE_CURRENT_SOURCE_DIR})
  endforeach()
endif()
add_executable(powerdrive-benchmark benchmarks/main.cpp)
target_link_libraries(powerdrive-benchmark PRIVATE powerdrive)
